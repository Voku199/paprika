```python
"""Čisté DSP funkce nad numpy/scipy. Žádná hudební teorie, žádný datový model."""
from __future__ import annotations
import math
import numpy as np
from scipy import signal

__all__ = ["KINDS", "biquad", "filt", "filt_varying", "res_to_q", "onepole",
           "lfo", "colored_noise", "karplus", "synth_ir", "blep_saw"]

KINDS = ("lp", "hp", "bp", "notch", "peak", "lowshelf", "highshelf")
Q_MIN, Q_MAX = 0.707, 12.0

def res_to_q(res: float) -> float:
    r = float(np.clip(res, 0.0, 1.0))
    return Q_MIN * (Q_MAX / Q_MIN) ** r

def biquad(kind, cutoff, sr, q=0.707, gain_db=0.0):
    if kind not in KINDS:
        raise ValueError(f"neznámý typ filtru {kind!r}; k dispozici: {', '.join(KINDS)}")
    f = float(np.clip(cutoff, 10.0, sr * 0.49))
    q = max(1e-3, float(q))
    w0 = 2.0 * math.pi * f / sr
    cw, sw = math.cos(w0), math.sin(w0)
    alpha = sw / (2.0 * q)
    A = 10.0 ** (gain_db / 40.0)
    if kind == "lp":
        b = [(1 - cw) / 2, 1 - cw, (1 - cw) / 2]; a = [1 + alpha, -2 * cw, 1 - alpha]
    elif kind == "hp":
        b = [(1 + cw) / 2, -(1 + cw), (1 + cw) / 2]; a = [1 + alpha, -2 * cw, 1 - alpha]
    elif kind == "bp":
        b = [alpha, 0.0, -alpha]; a = [1 + alpha, -2 * cw, 1 - alpha]
    elif kind == "notch":
        b = [1.0, -2 * cw, 1.0]; a = [1 + alpha, -2 * cw, 1 - alpha]
    elif kind == "peak":
        b = [1 + alpha * A, -2 * cw, 1 - alpha * A]; a = [1 + alpha / A, -2 * cw, 1 - alpha / A]
    else:
        sq = 2.0 * math.sqrt(A) * alpha
        if kind == "lowshelf":
            b = [A * ((A + 1) - (A - 1) * cw + sq), 2 * A * ((A - 1) - (A + 1) * cw),
                 A * ((A + 1) - (A - 1) * cw - sq)]
            a = [(A + 1) + (A - 1) * cw + sq, -2 * ((A - 1) + (A + 1) * cw),
                 (A + 1) + (A - 1) * cw - sq]
        else:
            b = [A * ((A + 1) + (A - 1) * cw + sq), -2 * A * ((A - 1) + (A + 1) * cw),
                 A * ((A + 1) + (A - 1) * cw - sq)]
            a = [(A + 1) - (A - 1) * cw + sq, 2 * ((A - 1) - (A + 1) * cw),
                 (A + 1) - (A - 1) * cw - sq]
    b = np.asarray(b) / a[0]; a = np.asarray(a) / a[0]
    return b, a

def filt(x, kind, cutoff, sr, q=0.707, gain_db=0.0, stages=1):
    b, a = biquad(kind, cutoff, sr, q, gain_db)
    y = np.asarray(x, dtype=np.float64)
    for _ in range(max(1, int(stages))):
        y = signal.lfilter(b, a, y, axis=0)
    return y.astype(np.float32)

def _zi_shape(order, x):
    return (order,) + tuple(x.shape[1:])

def filt_varying(x, kind, cutoff, sr, q=0.707, gain_db=0.0, stages=1, block=256):
    x = np.asarray(x, dtype=np.float64)
    n = x.shape[0]
    if np.isscalar(cutoff) and np.isscalar(q):
        return filt(x, kind, cutoff, sr, q, gain_db, stages)
    cut = np.broadcast_to(np.asarray(cutoff, dtype=np.float64), (n,))
    qq = np.broadcast_to(np.asarray(q, dtype=np.float64), (n,))
    y = x
    for _ in range(max(1, int(stages))):
        out = np.empty_like(y)
        zi = np.zeros(_zi_shape(2, y))
        for s in range(0, n, block):
            e = min(n, s + block); m = (s + e) // 2
            b, a = biquad(kind, cut[m], sr, qq[m], gain_db)
            out[s:e], zi = signal.lfilter(b, a, y[s:e], axis=0, zi=zi)
        y = out
    return y.astype(np.float32)

def onepole(x, cutoff, sr, poles=1, block=64):
    """Jednopólový LP (6 dB/okt na pól). Konstantní i proměnný cutoff přes lfilter."""
    x = np.asarray(x, dtype=np.float64)
    if x.shape[0] == 0:
        return x.astype(np.float32)
    def coef(c):
        c = float(np.clip(c, 10.0, sr * 0.49))
        return 1.0 - math.exp(-2.0 * math.pi * c / sr)
    y = x
    for _ in range(max(1, int(poles))):
        if np.isscalar(cutoff):
            g = coef(cutoff)
            y = signal.lfilter([g], [1.0, g - 1.0], y, axis=0)
            continue
        cut = np.asarray(cutoff, dtype=np.float64)
        out = np.empty_like(y); zi = np.zeros(_zi_shape(1, y))
        for s in range(0, y.shape[0], block):
            e = min(y.shape[0], s + block); g = coef(cut[(s + e) // 2])
            out[s:e], zi = signal.lfilter([g], [1.0, g - 1.0], y[s:e], axis=0, zi=zi)
        y = out
    return y.astype(np.float32)
```

```python
LFO_SHAPES = ("sine", "tri", "saw", "square", "random")

def lfo(shape, rate_hz, n, sr, phase=0.0, seed=0):
    if shape not in LFO_SHAPES:
        raise ValueError(f"neznámý tvar LFO {shape!r}; k dispozici: {', '.join(LFO_SHAPES)}")
    cyc = float(rate_hz) * np.arange(n, dtype=np.float64) / sr + float(phase)
    ph = cyc % 1.0
    if shape == "sine": return np.sin(2 * np.pi * ph)
    if shape == "tri": return 4.0 * np.abs(ph - 0.5) - 1.0
    if shape == "saw": return 2.0 * ph - 1.0
    if shape == "square": return np.where(ph < 0.5, 1.0, -1.0)
    idx = np.floor(cyc).astype(np.int64); idx -= idx.min() if n else 0
    vals = np.random.default_rng(seed).uniform(-1, 1, int(idx.max()) + 2 if n else 1)
    held = vals[idx]
    smooth = onepole(held, max(0.5, rate_hz * 4.0), sr).astype(np.float64)
    return np.clip(smooth, -1.0, 1.0)

_PINK_B = [0.049922035, -0.095993537, 0.050612699, -0.004408786]
_PINK_A = [1.0, -2.494956002, 2.017265875, -0.522189400]

def colored_noise(n, color="white", seed=0):
    x = np.random.default_rng(seed).uniform(-1.0, 1.0, n)
    if color == "white": y = x
    elif color == "pink": y = signal.lfilter(_PINK_B, _PINK_A, x)
    elif color == "brown": y = signal.lfilter([1.0], [1.0, -0.995], x); y -= y.mean()
    else: raise ValueError(f"neznámá barva šumu {color!r}; k dispozici: white, pink, brown")
    peak = np.abs(y).max() if n else 0.0
    return (y / peak if peak > 0 else y).astype(np.float32)

def karplus(freq, n, sr, decay=1.0, bright=0.5, seed=0):
    period = sr / max(20.0, float(freq))
    D = max(1, int(math.floor(period - 0.5))); frac = period - 0.5 - D
    C = (1.0 - frac) / (1.0 + frac)
    g = min(0.99999, 10.0 ** (-3.0 * (period / sr) / max(1e-3, decay)))
    a = np.zeros(D + 3); a[0] += 1.0; a[1] += C
    a[D] -= g / 2 * C; a[D + 1] -= g / 2 * (1 + C); a[D + 2] -= g / 2
    b = np.array([1.0, C])
    burst = np.zeros(n)
    m = min(n, int(math.ceil(period)))
    exc = np.random.default_rng(seed).uniform(-1, 1, m)
    exc = onepole(exc, 300.0 + float(np.clip(bright, 0, 1)) * (sr * 0.45 - 300.0), sr).astype(np.float64)
    burst[:m] = exc - exc.mean()
    y = signal.lfilter(b, a, burst)
    peak = np.abs(y).max()
    return (y / peak if peak > 0 else y).astype(np.float32)

def synth_ir(seconds, sr, predelay=0.02, damp=0.5, early=0.3, seed=0):
    seconds = max(0.05, float(seconds))
    m = int(seconds * 1.1 * sr); t = np.arange(m) / sr
    env = np.exp(-6.9078 * t / seconds)
    rng = np.random.default_rng(seed)
    out = np.zeros((m, 2))
    for ch in range(2):
        nz = rng.standard_normal(m)
        dark = onepole(nz, 1500.0, sr, poles=2).astype(np.float64)
        dark *= np.std(nz) / max(1e-9, np.std(dark))
        w = np.clip(t / seconds, 0, 1) ** 0.5 * float(np.clip(damp, 0, 1))
        tail = ((1 - w) * nz + w * dark) * env
        if early > 0:
            for k in range(8):
                i = int(rng.uniform(0.003, 0.08) * sr)
                tail[i] += rng.choice([-1, 1]) * early * 3.0 * (0.8 ** k)
        out[:, ch] = tail / np.sqrt((tail ** 2).sum())
    pre = np.zeros((int(max(0.0, predelay) * sr), 2))
    return np.concatenate([pre, out]).astype(np.float32)

def blep_saw(cycles, dt):
    t = np.asarray(cycles, dtype=np.float64) % 1.0
    dt = np.broadcast_to(np.clip(np.asarray(dt, dtype=np.float64), 1e-6, 0.5), t.shape)
    y = 2.0 * t - 1.0
    lo = t < dt; x = t[lo] / dt[lo]; y[lo] -= x + x - x * x - 1.0
    hi = t > 1.0 - dt; x = (t[hi] - 1.0) / dt[hi]; y[hi] -= x * x + x + x + 1.0
    return y
```

```python
class Osc:
    def __init__(self, ratio=1.0, amp=1.0, phase=0.0): ...
    @property
    def stereo(self) -> bool: return False
    def _cycles(self, freq, n, sr, phase=0.0):
        ph = self.phase + float(phase)
        if np.isscalar(freq):
            return float(freq) * self.ratio * np.arange(n, dtype=np.float64) / sr + ph
        f = np.asarray(freq, dtype=np.float64)[:n] * self.ratio
        return np.cumsum(f) / sr - f[0] / sr + ph
    @staticmethod
    def _fmax(freq, ratio): return float(np.max(freq)) * ratio

class Saw(Osc):
    def render(self, freq, n, sr, phase=0.0):
        f = self._fmax(freq, self.ratio)
        if f <= 0: return np.zeros(n, np.float32)
        cyc = 2 * np.pi * self._cycles(freq, n, sr, phase)
        out = np.zeros(n)
        for k in range(1, max(1, min(64, int(sr * 0.5 / f))) + 1):
            out -= np.sin(k * cyc) / k
        return (out * 2 / np.pi).astype(np.float32)
```

```python
def voice(self, freq, dur, vel=100, sr=44100, bpm=120.0, rr_index=0, tone_vel=None,
          glide_path=None, glide_time=0.0):
    gate_n = max(1, int(round(dur * sr)))
    total_n = gate_n + max(1, int(round(self.adsr.r * sr))) + 1
    semis = np.zeros(total_n); moved = False
    if self.pitch_env is not None: semis += self.pitch_env.curve(total_n, sr); moved = True
    if self.vibrato is not None: semis += self.vibrato.curve(total_n, sr, bpm); moved = True
    if self.drift: semis += self.drift / 100.0 * dsp.lfo("random", 0.7, total_n, sr, seed=rr_index + 17); moved = True
    rr = max(1, int(self.rr)); phase = 0.0
    if rr > 1:
        semis += RR_CENTS[rr_index % rr] / 100.0; phase = (rr_index * 0.37) % 1.0; moved = True
    f = _glide_curve(glide_path, glide_time, total_n, sr) if glide_path else float(freq)
    if moved:
        f = np.broadcast_to(np.asarray(f, dtype=np.float64), (total_n,)) * 2.0 ** (semis / 12.0)
    parts = [o.amp * np.asarray(o.render(f, total_n, sr, phase=phase), np.float32) for o in self.osc]
    stereo = any(p.ndim == 2 for p in parts)
    buf = np.zeros((total_n, 2) if stereo else total_n, np.float32)
    for p in parts: buf += p[:, None] if (stereo and p.ndim == 1) else p
    env = self.adsr.render(gate_n, total_n, sr)
    if self.tremolo is not None:
        env = env * (1.0 - self.tremolo.depth * (0.5 + 0.5 * self.tremolo.curve(total_n, sr, bpm) / max(1e-9, self.tremolo.depth))).astype(np.float32)
    buf *= env[:, None] if stereo else env
    tv = vel if tone_vel is None else tone_vel

    base_f = float(freq if glide_path is None else glide_path[-1][1])
    for flt in _as_list(self.filter):
```
