"""SFZ: text nástroje -> seznam regionů.

Modul zná jen text a čísla, žádné audio. Umí podmnožinu SFZ, která stačí na
běžné knihovny (Salamander, VSCO, vlastní WAV): hlavičky `<control>
<global> <master> <group> <region>`, `#define`, `#include`, komentáře
a cesty ke vzorkům s mezerami.
"""

from __future__ import annotations

import os
import re
from dataclasses import dataclass

__all__ = ["Region", "SfzError", "parse", "load", "note_number"]


class SfzError(ValueError):
    """SFZ se nedá přečíst — zpráva říká soubor, opcode a hodnotu."""


@dataclass
class Region:
    """Jeden region: vzorek a kdy/jak se hraje. Časy v sekundách."""

    sample: str
    lokey: int = 0
    hikey: int = 127
    keycenter: int = 60
    lovel: int = 1
    hivel: int = 127
    seq_length: int = 1
    seq_position: int = 1
    tune: float = 0.0          # centy
    transpose: int = 0         # půltóny
    volume: float = 0.0        # dB
    pan: float = 0.0           # -100 (vlevo) .. 100 (vpravo)
    veltrack: float = 100.0    # % (amp_veltrack)
    attack: float = 0.0
    release: float = 0.001
    offset: int = 0            # vzorky od začátku souboru
    loop_mode: str = "no_loop"
    loop_start: int = 0
    loop_end: int = 0
    trigger: str = "attack"
    ignored: tuple = ()        # opcody, kterým parser nerozumí


HEADERS = ("control", "global", "master", "group", "region")
_ALIASES = {"loopmode": "loop_mode", "loopstart": "loop_start",
            "loopend": "loop_end"}
_INT = {"lovel": "lovel", "hivel": "hivel", "seq_length": "seq_length",
        "seq_position": "seq_position", "transpose": "transpose",
        "offset": "offset", "loop_start": "loop_start", "loop_end": "loop_end"}
_FLOAT = {"tune": "tune", "volume": "volume", "pan": "pan",
          "amp_veltrack": "veltrack", "ampeg_attack": "attack",
          "ampeg_release": "release"}
_NOTE_RE = re.compile(r"^([a-gA-G])([#b]?)(-?\d+)$")
_PCS = {"c": 0, "d": 2, "e": 4, "f": 5, "g": 7, "a": 9, "b": 11}
_TOKEN = re.compile(r"<(\w+)>|([A-Za-z0-9_]+)=")
_DEFINE = re.compile(r"^\s*#define\s+(\$\w+)\s+(.*?)\s*$")
_INCLUDE = re.compile(r'^\s*#include\s+"([^"]+)"\s*$')


def note_number(value) -> int:
    """'60', 'c4', 'F#3', 'eb2' -> MIDI číslo (c4 = 60)."""
    text = str(value).strip()
    try:
        return int(text)
    except ValueError:
        pass
    m = _NOTE_RE.match(text)
    if not m:
        raise SfzError(f"neplatná nota {value!r} (čekám číslo nebo např. c4, f#3)")
    pc = _PCS[m.group(1).lower()] + {"#": 1, "b": -1, "": 0}[m.group(2)]
    return (int(m.group(3)) + 1) * 12 + pc


def _read(path: str) -> str:
    try:
        with open(path, "rb") as f:
            raw = f.read()
    except OSError as e:
        raise SfzError(f"SFZ soubor se nedá otevřít: {path} ({e.strerror})") from e
    try:
        return raw.decode("utf-8")
    except UnicodeDecodeError:
        return raw.decode("latin-1")


def _strip_comments(text: str) -> str:
    text = re.sub(r"/\*.*?\*/", " ", text, flags=re.S)
    return re.sub(r"//[^\n]*", "", text)


def _expand(text: str, base_dir: str, depth: int = 0) -> str:
    """`#include` a `#define` -> čistý text."""
    if depth > 8:
        raise SfzError("#include je zanořený příliš hluboko (cyklus?)")
    defines: dict = {}
    out = []
    for line in text.splitlines():
        m = _DEFINE.match(line)
        if m:
            defines[m.group(1)] = m.group(2)
            continue
        m = _INCLUDE.match(line)
        if m:
            path = os.path.join(base_dir, m.group(1).replace("\\", "/"))
            out.append(_expand(_strip_comments(_read(path)), base_dir, depth + 1))
            continue
        for name in sorted(defines, key=len, reverse=True):
            line = line.replace(name, defines[name])
        out.append(line)
    return "\n".join(out)


def _tokens(text: str):
    """Text -> ('header', jméno, '') a ('op', jméno, hodnota).

    Hodnota sahá až k dalšímu `opcode=` nebo hlavičce na řádku, takže
    `sample=` smí obsahovat mezery.
    """
    for line in text.splitlines():
        found = list(_TOKEN.finditer(line))
        for i, m in enumerate(found):
            if m.group(1):
                yield "header", m.group(1).lower(), ""
                continue
            end = found[i + 1].start() if i + 1 < len(found) else len(line)
            yield "op", m.group(2).lower(), line[m.end():end].strip()


def _region(ops: dict, base_dir: str, default_path: str,
            source: str) -> Region | None:
    sample = ops.get("sample", "")
    if not sample or sample.startswith("*"):
        return None               # generátory (*sine, *silence) nepodporujeme
    rel = (default_path + sample).replace("\\", "/")
    r = Region(sample=os.path.normpath(os.path.join(base_dir, rel)))
    ignored = []
    # `key` nastaví i keycenter, takže musí přijít před pitch_keycenter
    for name, value in sorted(ops.items(), key=lambda kv: kv[0] != "key"):
        try:
            if name == "sample":
                continue
            if name == "key":
                r.lokey = r.hikey = r.keycenter = note_number(value)
            elif name == "pitch_keycenter":
                if value.lower() != "sample":
                    r.keycenter = note_number(value)
            elif name in ("lokey", "hikey"):
                setattr(r, name, note_number(value))
            elif name in _INT:
                setattr(r, _INT[name], int(float(value.split()[0])))
            elif name in _FLOAT:
                setattr(r, _FLOAT[name], float(value.split()[0]))
            elif name in ("loop_mode", "trigger"):
                setattr(r, name, value.split()[0].lower())
            else:
                ignored.append(name)
        except (ValueError, IndexError) as e:
            raise SfzError(f"{source}: opcode {name}={value!r} má neplatnou "
                           f"hodnotu") from e
    r.ignored = tuple(sorted(ignored))
    return r


def parse(text: str, base_dir: str = ".", source: str = "<sfz>") -> list[Region]:
    """Text SFZ -> regiony s cestami ke vzorkům složenými s `base_dir`."""
    text = _expand(_strip_comments(text), base_dir)
    scopes = {h: {} for h in HEADERS}
    current = None
    regions: list[Region] = []

    def close():
        if current != "region":
            return
        merged: dict = {}
        for h in ("global", "master", "group", "region"):
            merged.update(scopes[h])
        r = _region(merged, base_dir, scopes["control"].get("default_path", ""),
                    source)
        if r is not None:
            regions.append(r)

    resets = {"control": ("control",), "global": ("global", "master", "group"),
              "master": ("master", "group"), "group": ("group",),
              "region": ("region",)}
    for kind, name, value in _tokens(text):
        if kind == "header":
            close()
            current = name if name in HEADERS else None
            for h in resets.get(current, ()):
                scopes[h] = {}
            continue
        if current is not None:
            scopes[current][_ALIASES.get(name, name)] = value
    close()
    return regions


def load(path: str) -> list[Region]:
    """SFZ soubor -> regiony (cesty ke vzorkům relativně k souboru)."""
    base = os.path.dirname(os.path.abspath(path))
    return parse(_read(path), base, source=os.path.basename(path))

"""Malé syntetické knihovny vzorků pro testy (sinusovky, žádná síť)."""

import os

import numpy as np
from scipy.io import wavfile


def midi_freq(p):
    return 440.0 * 2.0 ** ((p - 69) / 12.0)


def sine_wav(path, freq, sr=44100, seconds=1.0, amp=0.5, stereo=True):
    t = np.arange(int(sr * seconds)) / sr
    x = amp * np.sin(2 * np.pi * freq * t)
    if stereo:
        x = np.stack([x, x], axis=1)
    os.makedirs(os.path.dirname(path), exist_ok=True)
    wavfile.write(path, sr, (x * 32767).astype(np.int16))
    return path


def make_sfz_library(root, sr=48000, seconds=1.0):
    """Kořeny C4/E4/G#4, dvě velocity vrstvy, u C4 dva round robiny.

    Vzorky leží ve složce s mezerou ve jménu. Vrací cestu k SFZ.
    """
    lines = ["<global> amp_veltrack=100 ampeg_release=0.2"]
    for root_p, lo, hi in ((60, 58, 62), (64, 63, 66), (68, 67, 70)):
        for layer, (lovel, hivel, amp) in enumerate(((1, 63, 0.2), (64, 127, 0.8)), 1):
            rrs = (1, 2) if root_p == 60 else (1,)
            for rr in rrs:
                name = f"s_{root_p}_v{layer}_rr{rr}.wav"
                sine_wav(os.path.join(root, "wav dir", name), midi_freq(root_p),
                         sr, seconds, amp * (1.0 - 0.1 * (rr - 1)))
                lines.append(
                    f"<region> sample=wav dir/{name} lokey={lo} hikey={hi} "
                    f"pitch_keycenter={root_p} lovel={lovel} hivel={hivel} "
                    f"seq_length={len(rrs)} seq_position={rr}")
    path = os.path.join(root, "test.sfz")
    with open(path, "w", encoding="utf-8") as f:
        f.write("\n".join(lines) + "\n")
    return path


VSCO_VIOLINS = {
    "Strings/Violin Section/susVib": "VlnEns_susVib_{n}_v{v}.wav",
    "Strings/Violin Section/Trem": "VlnEns_Trem_{n}_v{v}.wav",
    "Strings/Violin Section/Pizz": "VlnEns_Pizz_{n}_v{v}_rr{r}.wav",
    "Strings/Violin Section/Spic": "VlnEns_Spic_{n}_v{v}_rr{r}.wav",
}


def make_vsco_violins(root, notes=("A2", "C3", "E3")):
    """Složky VSCO pro `violins` se jmény jako ve skutečném repu.

    VSCO jmenuje s C3 = střední C, takže A2 zní jako MIDI 57.
    """
    pcs = {"C": 0, "D": 2, "E": 4, "F": 5, "G": 7, "A": 9, "B": 11}
    for folder, pattern in VSCO_VIOLINS.items():
        for n in notes:
            pitch = (int(n[-1]) + 1) * 12 + pcs[n[0]] + 12
            for v in (1, 2):
                for r in (1, 2):
                    name = pattern.format(n=n, v=v, r=r)
                    sine_wav(os.path.join(root, "vsco2", *folder.split("/"), name),
                             midi_freq(pitch), 44100, 0.6, 0.15 * v)

SR = 44100


def peak_freq(x, sr=SR):
    m = x.mean(axis=1) if x.ndim == 2 else x
    size = 1 << 15
    m = m[:size] if m.size >= size else np.pad(m, (0, size - m.size))
    spec = np.abs(np.fft.rfft(m * np.hanning(size)))
    k = int(np.argmax(spec))
    a, b, c = np.log(spec[k - 1:k + 2] + 1e-12)
    return (k + 0.5 * (a - c) / (a - 2 * b + c)) * sr / size


def write(path, text):

"""Sampler: skutečné nástroje z nahrávek (SFZ + WAV).

Nezná skladbu, beaty ani pedál — dostane výšku, velocity a délku v sekundách
a vrátí stereo hlas `(n, 2)`. Rozvrh not (pedál, legato) dělá `render.py`.
WAV se čte líně přes mmap, takže velká knihovna nestojí paměť.
"""

from __future__ import annotations

import gc
import math
import os
import warnings

import numpy as np
from scipy.io import wavfile

from . import sfz as _sfz

__all__ = ["Sampler", "SampleError", "load_wav", "to_float", "clear_cache",
           "LEGATO_OFFSET", "LEGATO_XFADE", "TAIL_CAP", "UNDAMPED_RELEASE"]

LEGATO_OFFSET = 0.08      # s: legato nota začíná za nasazením smyčce / jazyku
LEGATO_XFADE = 0.08       # s: prolnutí dvou legato not
EDGE_FADE = 0.002         # s: fade na krajích hlasu, jinak to lupne
UNDAMPED_RELEASE = 2.0    # s: struny bez dusítek (nejvyšší oktávy piana)
TAIL_CAP = 8.0            # s: strop doznění pro délku bufferu
LOOPING = ("loop_continuous", "loop_sustain")

_WAVS: dict = {}          # cesta -> (sr, data); data je mmap, žádná kopie


class SampleError(RuntimeError):
    """Vzorek nebo SFZ chybí či je vadný — zpráva obsahuje cestu."""


def load_wav(path: str):
    """WAV -> `(sr, data (n, kanály))`. Mmap, když to formát dovolí; cache."""
    hit = _WAVS.get(path)
    if hit is not None:
        return hit
    if not os.path.isfile(path):
        raise SampleError(f"vzorek neexistuje: {path}")
    with warnings.catch_warnings():
        warnings.simplefilter("ignore")          # neznámé chunky (smpl, LIST)
        try:
            try:
                sr, data = wavfile.read(path, mmap=True)
            except ValueError:                   # 24 bit mmap neumí
                sr, data = wavfile.read(path)
        except Exception as e:  # noqa: BLE001 - scipy hází různé typy
            raise SampleError(f"vzorek se nedá přečíst: {path} ({e})") from e
    if data.ndim == 1:
        data = data[:, None]
    _WAVS[path] = (int(sr), data)
    return _WAVS[path]


def clear_cache() -> None:
    """Zavře mmapy vzorků (Windows jinak drží soubory otevřené)."""
    _WAVS.clear()
    gc.collect()


def to_float(block: np.ndarray) -> np.ndarray:
    """Blok PCM -> float32 stereo `(m, 2)` v rozsahu -1..1."""
    if block.dtype == np.int16:
        out = block.astype(np.float32) / 32768.0
    elif block.dtype == np.int32:                # i 24 bit (zarovnaný nahoru)
        out = block.astype(np.float32) / 2147483648.0
    elif block.dtype == np.uint8:
        out = (block.astype(np.float32) - 128.0) / 128.0
    else:
        out = block.astype(np.float32)
    if out.shape[1] == 1:
        return np.repeat(out, 2, axis=1)
    return out[:, :2]


def _read(src: np.ndarray, pos: np.ndarray) -> np.ndarray:
    """Hodnoty `src` v neceločíselných pozicích (4bodový Hermite). Mimo = 0."""
    out = np.zeros((pos.size, 2), dtype=np.float32)
    n = src.shape[0]
    if pos.size == 0 or n == 0:
        return out
    idx = np.floor(pos).astype(np.int64)
    ok = (idx >= 0) & (idx < n)
    if not np.any(ok):
        return out
    lo = max(0, int(idx[ok].min()) - 1)
    hi = min(n, int(idx[ok].max()) + 3)
    pad = np.zeros((2, 2), np.float32)
    block = np.concatenate([pad, to_float(src[lo:hi]), pad])
    j = idx[ok] - lo + 2
    f = (pos[ok] - idx[ok]).astype(np.float32)[:, None]
    y0, y1, y2, y3 = block[j - 1], block[j], block[j + 1], block[j + 2]
    c1 = 0.5 * (y2 - y0)
    c2 = y0 - 2.5 * y1 + 2.0 * y2 - 0.5 * y3
    c3 = 0.5 * (y3 - y0) + 1.5 * (y1 - y2)
    out[ok] = ((c3 * f + c2) * f + c1) * f + y1
    return out


def _looped(data: np.ndarray, start: int, end: int, need: int,
            sr: int) -> np.ndarray:
    """Vzorek jako float, smyčka `[start, end)` opakovaná na ≥ `need` vzorků.

    Konec smyčky se prolne (rovný výkon) s tím, co jí předchází, takže skok
    zpět na `start` je plynulý.
    """
    x = to_float(data[:end])
    length = end - start
    xf = int(min(0.25 * length, 0.5 * sr, start))
    seg = x[start:end].copy()
    if xf > 0:
        t = np.linspace(0.0, 1.0, xf, dtype=np.float32)[:, None]
        seg[-xf:] = (seg[-xf:] * np.cos(t * np.pi / 2)
                     + x[start - xf:start] * np.sin(t * np.pi / 2))
    reps = max(1, math.ceil((need - start) / length) + 1)
    return np.concatenate([x[:start]] + [seg] * reps)


def _fall(x: np.ndarray) -> np.ndarray:
    """Exponenciální pokles 1 -> 0 na x ∈ [0, 1], přesně 0 v x = 1."""
    x = np.clip(x, 0.0, 1.0).astype(np.float64)
    y = (np.exp(-5.0 * x) - math.exp(-5.0)) / (1.0 - math.exp(-5.0))
    return y.astype(np.float32)


class Sampler:
    """Nahraný nástroj z SFZ. Ve stopě se používá místo `Patch`.

    `legato` říká, jestli artikulace umí legato (výdrže ano, pizz ne);
    `undamped_from` je nejnižší MIDI výška bez dusítek (piano ~89);
    `gain_db` srovná hlasitost knihovny s ostatními nástroji.
    """

    sampled = True

    def __init__(self, sfz_path: str, name: str | None = None,
                 legato: bool = False, undamped_from: int | None = None,
                 gain_db: float = 0.0):
        self.path = os.path.abspath(sfz_path)
        if not os.path.isfile(self.path):
            raise SampleError(f"SFZ neexistuje: {self.path}")
        try:
            regions = _sfz.load(self.path)
        except _sfz.SfzError as e:
            raise SampleError(str(e)) from e
        self.regions = [r for r in regions if r.trigger in ("attack", "first")]
        if not self.regions:
            raise SampleError(f"SFZ nemá žádné hratelné regiony: {self.path}")
        self.name = name or os.path.splitext(os.path.basename(self.path))[0]
        self.legato = bool(legato)
        self.undamped_from = undamped_from
        self.gain_db = float(gain_db)
        self._ids = {id(r): i for i, r in enumerate(self.regions)}
        self._mtime = os.path.getmtime(self.path)

    def __repr__(self) -> str:  # pragma: no cover
        return f"<Sampler {self.name} regions={len(self.regions)}>"

    @property
    def range(self) -> tuple[int, int]:
        """Nejnižší a nejvyšší klávesa, kterou knihovna pokrývá."""
        return (min(r.lokey for r in self.regions),
                max(r.hikey for r in self.regions))

    @property
    def tail(self) -> float:
        """Nejdelší doznění po konci noty (pro délku bufferu)."""
        rel = max(r.release for r in self.regions)
        if self.undamped_from is not None:
            rel = max(rel, UNDAMPED_RELEASE)
        if any(r.loop_mode == "one_shot" for r in self.regions):
            rel = TAIL_CAP
        return float(min(TAIL_CAP, rel))

    def key(self) -> tuple:
        return ("sampler", self.path, self._mtime, self.undamped_from,
                self.gain_db)

    def region_id(self, region) -> int:
        return self._ids[id(region)]

    def select(self, pitch: int, vel: int, rr_index: int = 0) -> list:
        """Regiony, které pro notu zazní.

        Round robin (`seq_length` > 1) vybírá podle `rr_index`; ostatní
        sedící regiony zní spolu (vrstvy). Mimo rozsah vezme region
        s nejbližším kořenem.
        """
        vel = max(1, min(127, int(vel)))
        by_vel = [r for r in self.regions if r.lovel <= vel <= r.hivel] or self.regions
        cands = [r for r in by_vel if r.lokey <= pitch <= r.hikey]
        if not cands:
            best = min(abs(r.keycenter - pitch) for r in by_vel)
            cands = [r for r in by_vel if abs(r.keycenter - pitch) == best]
        rr = [r for r in cands if r.seq_length > 1]
        if not rr:
            return cands
        pos = rr_index % rr[0].seq_length + 1
        chosen = [r for r in rr if r.seq_position == pos]
        if not chosen:          # díra v sekvenci: vezmi dostupné cyklicky
            ordered = sorted(rr, key=lambda r: (r.seq_position, r.sample))
            chosen = [ordered[rr_index % len(ordered)]]
        return chosen + [r for r in cands if r.seq_length <= 1]

    def gain(self, region, vel: int) -> float:
        """Zisk noty: SFZ zákon velocity (`amp_veltrack`) + hlasitost."""
        v = max(1, min(127, int(vel))) / 127.0
        db = (region.volume + self.gain_db
              + region.veltrack / 100.0 * 40.0 * math.log10(v))
        return float(10.0 ** (db / 20.0))

    def voice(self, region, pitch: int, dur: float, sr: int,
              legato_in: bool = False, legato_out: bool = False) -> np.ndarray:
        """Jedna nota z regionu -> stereo `(n, 2)` float32, bez velocity zisku.

        Posun výšky a převzorkování na `sr` jsou jeden poměr čtení. Po gate
        (`dur`) přijde release; `one_shot` přehraje vzorek celý.
        """
        src_sr, data = load_wav(region.sample)
        rate = (2.0 ** ((pitch - region.keycenter + region.transpose) / 12.0
                        + region.tune / 1200.0) * src_sr / sr)
        start = region.offset + (int(LEGATO_OFFSET * src_sr) if legato_in else 0)
        start = min(start, max(0, data.shape[0] - 1))
        one_shot = region.loop_mode == "one_shot"
        looping = (region.loop_mode in LOOPING
                   and region.loop_end > region.loop_start)
        gate_n = max(1, int(round(dur * sr)))
        if legato_out:
            rel = LEGATO_XFADE
        elif self.undamped_from is not None and pitch >= self.undamped_from:
            rel = max(region.release, UNDAMPED_RELEASE)
        else:
            rel = region.release
        rel_n = max(1, int(round(rel * sr)))
        avail = int((data.shape[0] - start) / rate)
        if one_shot:
            n_out = avail
        else:
            n_out = gate_n + rel_n
            if not looping:
                n_out = min(n_out, avail)
        n_out = max(1, n_out)
        pos = start + np.arange(n_out, dtype=np.float64) * rate
        src = data
        if looping and pos[-1] >= region.loop_end - 1:
            end = min(region.loop_end, data.shape[0])
            src = _looped(data, min(region.loop_start, end - 1), end,
                          int(pos[-1]) + 4, src_sr)
        out = _read(src, pos)
        out *= self._envelope(n_out, gate_n, rel_n, sr, region.attack, one_shot,
                              legato_in, legato_out)[:, None]
        if region.pan:
            p = max(-1.0, min(1.0, region.pan / 100.0))
            out[:, 0] *= min(1.0, 1.0 - p)
            out[:, 1] *= min(1.0, 1.0 + p)
        return out

    @staticmethod
    def _envelope(n: int, gate_n: int, rel_n: int, sr: int, attack: float,
                  one_shot: bool, legato_in: bool, legato_out: bool) -> np.ndarray:
        env = np.ones(n, dtype=np.float32)
        a_n = min(n, int(max(attack, LEGATO_XFADE if legato_in else 0.0,
                             EDGE_FADE) * sr))
        if a_n > 0:
            x = np.linspace(0.0, 1.0, a_n, endpoint=False, dtype=np.float32)
            env[:a_n] = np.sin(x * np.pi / 2) if legato_in else x
        if not one_shot and n > gate_n:
            x = np.arange(n - gate_n, dtype=np.float32) / rel_n
            fall = (np.cos(np.clip(x, 0.0, 1.0) * np.pi / 2) if legato_out
                    else _fall(x))
            env[gate_n:] *= fall
        e_n = min(n, int(EDGE_FADE * sr))
        if e_n > 0:
            env[n - e_n:] *= np.linspace(1.0, 0.0, e_n, dtype=np.float32)
        return env

"""Instalace vzorků: stažení, rozbalení, vygenerování SFZ.

Jediný modul Paprika, který sahá na síť. Stahování pokračuje: hotové
soubory se správnou velikostí se přeskočí, rozstahované leží jako `.part`.
"""

from __future__ import annotations

import math
import os
import re

import numpy as np

from .library import Instrument
from .sampler import load_wav, to_float
from .sfz import note_number

__all__ = ["InstallError", "parse_sample_name", "plan_regions", "write_sfz",
           "VSCO_SHA", "VSCO_OCTAVE_ZERO", "TARGET_RMS_DB"]

VSCO_SHA = "440300901dfe9275fd84e0b7763af1f8443ae62e"
TARGET_RMS_DB = -18.0     # max RMS nejhlasitější vrstvy prostředního kořene
# VSCO pojmenovává většinu složek s C3 = střední C (výška z jména + 12);
# tyhle používají C4 = 60. Změřeno detekcí výšky všech složek 2026-09-24.
VSCO_OCTAVE_ZERO = ("Keys/Upright Nr1", "Strings/Harp",
                    "Strings/Solo Violin/Arco Vib", "Strings/Solo Violin/Pizz")
LOOP_START, LOOP_END = 0.45, 0.90
_DYN = {"ppp": 1, "pp": 2, "p": 3, "mp": 4, "mf": 5, "f": 6, "ff": 7, "fff": 8,
        "soft": 3, "medium": 5, "loud": 7}
_NOTE = re.compile(r"^[A-G][#b]?-?\d$")


class InstallError(RuntimeError):
    """Instalace selhala — zpráva říká co a jak pokračovat."""


def parse_sample_name(filename: str, octave_shift: int = 0):
    """Jméno VSCO souboru -> `(výška, vrstva, rr)`, bez noty `None`.

    Tokeny oddělené `_`: nota (`A#3`), vrstva (`v2` nebo dynamika `mf`),
    round robin (`rr2`, `RR2` nebo koncové číslo `_1`). Před notou se
    ignoruje všechno (jméno nástroje, tvrdost paliček).
    """
    stem = os.path.splitext(os.path.basename(filename))[0]
    pitch, layer, rr = None, 0, 1
    for tok in stem.split("_"):
        low = tok.lower()
        if pitch is None:
            if _NOTE.match(tok):
                pitch = note_number(tok) + octave_shift
            continue
        if re.fullmatch(r"v\d+", low):
            layer = int(low[1:])
        elif re.fullmatch(r"rr\d+", low):
            rr = int(low[2:])
        elif low in _DYN:
            layer = _DYN[low]
        elif tok.isdigit():
            rr = int(tok)
    return None if pitch is None else (pitch, layer, rr)


def plan_regions(entries) -> list[dict]:
    """`[(cesta, výška, vrstva, rr)]` -> regiony (dict opcodů).

    Velocity se dělí rovnoměrně mezi vrstvy; klávesy vrstvy napůl mezi
    sousední kořeny, krajní o 2 půltóny dál.
    """
    layers = sorted({e[2] for e in entries})
    out = []
    for li, layer in enumerate(layers):
        lovel = 1 + (li * 127) // len(layers)
        hivel = ((li + 1) * 127) // len(layers)
        mine = [e for e in entries if e[2] == layer]
        roots = sorted({e[1] for e in mine})
        for k, root in enumerate(roots):
            lokey = roots[0] - 2 if k == 0 else (roots[k - 1] + root) // 2 + 1
            hikey = (roots[-1] + 2 if k == len(roots) - 1
                     else (root + roots[k + 1]) // 2)
            group = sorted((e for e in mine if e[1] == root),
                           key=lambda e: (e[3], e[0]))
            for pos, e in enumerate(group, start=1):
                out.append({"sample": e[0], "lokey": max(0, lokey),
                            "hikey": min(127, hikey), "pitch_keycenter": root,
                            "lovel": lovel, "hivel": hivel,
                            "seq_length": len(group), "seq_position": pos})
    return out


def _loudest_rms_db(regions: list, base: str) -> float:
    """Max RMS (100 ms okna) nejhlasitější vrstvy u prostředního kořene."""
    top = max(r["hivel"] for r in regions)
    layer = [r for r in regions if r["hivel"] == top]
    roots = sorted({r["pitch_keycenter"] for r in layer})
    mid = roots[len(roots) // 2]
    r = next(r for r in layer if r["pitch_keycenter"] == mid)
    sr, data = load_wav(os.path.join(base, *r["sample"].split("/")))
    x = to_float(data[: sr * 4]).mean(axis=1)
    win = max(1, int(0.1 * sr))
    n = (x.size // win) * win
    if n == 0:
        return TARGET_RMS_DB
    rms = float(np.sqrt(np.mean(x[:n].reshape(-1, win) ** 2, axis=1)).max())
    return 20.0 * math.log10(max(rms, 1e-6))


def write_sfz(inst: Instrument, art: str, root: str) -> str:
    """Vygeneruje `<root>/vsco2/sfz/<nástroj>.<art>.sfz` ze stažených WAV."""
    folder = inst.arts[art]
    base = os.path.join(root, "vsco2")
    src_dir = os.path.join(base, *folder.split("/"))
    shift = 0 if folder in VSCO_OCTAVE_ZERO else 12
    entries = []
    names = sorted(os.listdir(src_dir)) if os.path.isdir(src_dir) else []
    for fn in names:
        if not fn.lower().endswith(".wav"):
            continue
        parsed = parse_sample_name(fn, shift)
        if parsed:
            entries.append((f"{folder}/{fn}", *parsed))
    if not entries:
        raise InstallError(f"ve složce {src_dir} nejsou vzorky pro "
                           f"{inst.name}.{art} — spusť instalaci znovu")
    regions = plan_regions(entries)
    for r in regions:
        if inst.loops(art):
            n = load_wav(os.path.join(base, *r["sample"].split("/")))[1].shape[0]
            r["loop_mode"] = "loop_continuous"
            r["loop_start"] = int(n * LOOP_START)
            r["loop_end"] = int(n * LOOP_END)
        elif inst.is_one_shot(art):
            r["loop_mode"] = "one_shot"
    volume = TARGET_RMS_DB - _loudest_rms_db(regions, base)
    veltrack = 100 if len({r["lovel"] for r in regions}) == 1 else 40
    lines = ["// vygenerováno Paprikou z VSCO 2 CE (CC0) — neupravuj, přepíše se",
             "<control> default_path=../",
             f"<global> volume={volume:.2f} amp_veltrack={veltrack} "
             f"ampeg_release={inst.release:g}"]
    for r in regions:
        lines.append("<region> " + " ".join(f"{k}={v}" for k, v in r.items()))
    path = os.path.join(base, "sfz", f"{inst.name}.{art}.sfz")
    os.makedirs(os.path.dirname(path), exist_ok=True)
    with open(path, "w", encoding="utf-8") as f:
        f.write("\n".join(lines) + "\n")
    return path

def _urlopen(url: str, timeout: float = 60.0):
    req = urllib.request.Request(url, headers={"User-Agent": "paprika-sampler"})
    return urllib.request.urlopen(req, timeout=timeout)  # noqa: S310 - pevné https URL


def download(url: str, dst: str, size: int | None = None) -> bool:
    """Stáhne `url` do `dst` přes `.part`. False = soubor už je hotový."""
    if os.path.isfile(dst) and (size is None or os.path.getsize(dst) == size):
        return False
    os.makedirs(os.path.dirname(dst) or ".", exist_ok=True)
    part = dst + ".part"
    try:
        with _urlopen(url) as resp, open(part, "wb") as f:
            while True:
                chunk = resp.read(1 << 20)
                if not chunk:
                    break
                f.write(chunk)
    except OSError as e:              # URLError je podtřída OSError
        if os.path.exists(part):
            os.remove(part)
        raise InstallError(f"stažení selhalo: {url} ({e}) — spusť instalaci "
                           f"znovu, hotové soubory se přeskočí") from e
    got = os.path.getsize(part)
    if size is not None and got != size:
        os.remove(part)
        raise InstallError(f"stažený soubor má {got} B místo {size} B: {url} — "
                           f"spusť instalaci znovu")
    os.replace(part, dst)
    return True


def extract(archive: str, dest: str) -> None:
    """Rozbalí tar(.bz2) do `dest`; člen mimo `dest` nebo odkaz = chyba."""
    dest_abs = os.path.abspath(dest)
    os.makedirs(dest_abs, exist_ok=True)
    with tarfile.open(archive, "r:*") as tar:
        for m in tar:
            target = os.path.abspath(os.path.join(dest_abs, m.name))
            if (os.path.commonpath([dest_abs, target]) != dest_abs
                    or not (m.isfile() or m.isdir())):
                raise InstallError(f"archiv {archive} obsahuje nebezpečnou "
                                   f"položku: {m.name}")
            tar.extract(m, dest_abs, filter="data")


def _install_salamander(root: str, log) -> None:
    dest = os.path.join(root, "salamander")
    marker = os.path.join(dest, ".complete")
    if os.path.isfile(marker):
        log("piano: Salamander Grand Piano už je nainstalovaný")
        return

def vsco_files(folders) -> list[tuple[str, int]]:
    """WAV soubory VSCO repa přímo ve složkách `folders` (cesta, velikost)."""
    try:
        with _urlopen(VSCO_TREE) as resp:
            tree = json.load(resp)["tree"]
    except (OSError, ValueError, KeyError) as e:
        raise InstallError(f"seznam souborů VSCO 2 CE se nepodařilo získat ({e}) "
                           f"— zkus to znovu za chvíli (limit GitHub API)") from e
    want = set(folders)
    return [(x["path"], int(x["size"])) for x in tree
            if x.get("type") == "blob" and x["path"].lower().endswith(".wav")
            and posixpath.dirname(x["path"]) in want]


def _download_vsco(files, root: str, log) -> None:
    base = os.path.join(root, "vsco2")
    total = sum(s for _, s in files)
    step = max(1, len(files) // 10)

    def one(item):
        path, size = item
        download(VSCO_RAW + urllib.parse.quote(path),
                 os.path.join(base, *path.split("/")), size)
        return size

    done = 0
    with ThreadPoolExecutor(8) as ex:
        for i, size in enumerate(ex.map(one, files), 1):
            done += size
            if i % step == 0 or i == len(files):
                log(f"  {i}/{len(files)} souborů, "
                    f"{done / 1e6:.0f}/{total / 1e6:.0f} MB")


def _write_manifest(root: str, groups) -> None:
    path = os.path.join(root, MANIFEST)
    data: dict = {}
    if os.path.isfile(path):
        try:
            with open(path, encoding="utf-8") as f:
                data = json.load(f)
        except ValueError:
            data = {}
    today = datetime.date.today().isoformat()
    data.setdefault("groups", {})
    for g in groups:
        data["groups"][g] = {"installed": today}
    data["vsco2_commit"] = VSCO_SHA
    data["salamander"] = SALAMANDER_URL
    with open(path, "w", encoding="utf-8") as f:
        json.dump(data, f, indent=2, ensure_ascii=False)


def install(group: str, root: str | None = None, log=print) -> list[str]:
    """Nainstaluje skupinu (`piano`, `strings`, ..., `all`). Vrací nástroje."""
    groups = list(GROUPS) if group == "all" else [group]
    for g in groups:
        if g not in GROUPS:
            raise InstallError(f"neznámá skupina {g!r}; k dispozici: "
                               f"{', '.join(GROUPS)}, all")
    root = os.path.abspath(root or samples_root())
    os.makedirs(root, exist_ok=True)
    done: list[str] = []
    for g in groups:
        insts = [i for i in INSTRUMENTS.values() if i.group == g]
        if any(i.library == "salamander" for i in insts):
            _install_salamander(root, log)
        vsco = [i for i in insts if i.library == "vsco2"]
        if vsco:
            folders = sorted({f for i in vsco for f in i.arts.values()})
            log(f"{g}: zjišťuji soubory VSCO 2 CE ...")
            files = vsco_files(folders)
            log(f"{g}: {len(files)} souborů, "
                f"{sum(s for _, s in files) / 1e6:.0f} MB")
            _download_vsco(files, root, log)
            for inst in vsco:
                for art in inst.arts:
                    write_sfz(inst, art, root)
            log(f"{g}: SFZ vygenerováno")
        done += [i.name for i in insts]
    _write_manifest(root, groups)
    with open(os.path.join(root, "CREDITS.txt"), "w", encoding="utf-8") as f:
        f.write(CREDITS)
    return done

    def pedal(self, bars, every: float | None = None):
        """Sustain pedál (piano): nota, která skončí při sešlápnutí, zní do
        uvolnění. `bars=(od, do)` v taktech; `every` = přešlápnutí po tolika
        taktech (1 = každý takt, jak to dělá pianista při změně harmonie)::

            piano.pedal(bars=(0, 8), every=1)

        Pedál se sešlápne `PEDAL_LIFT` beatu po začátku úseku, takže nota
        končící přesně na hranici se nezachytí (noha je zrovna nahoře).
        """
        lo, hi = float(bars[0]), float(bars[1])
        if hi <= lo:
            raise ValueError(f"pedal: bars=(od, do) musí mít do > od, "
                             f"dostal jsem {bars!r}")
        step = float(every) if every else hi - lo
        if step <= 0:
            raise ValueError("pedal: every musí být kladné")
        b = lo
        while b < hi - 1e-9:
            e = min(hi, b + step)
            self.pedals.append((self._bar_to_beat(b) + PEDAL_LIFT,
                                self._bar_to_beat(e)))
            b = e
        self.pedals.sort()
        return self

def _pedaled(notes: list, pedals: list) -> list:
    """Noty prodloužené sustain pedálem (nové instance, vstup se nemění).

    Nota, která skončí při sešlápnutém pedálu, zní do jeho uvolnění. Znovu
    udeřená stejná výška utne předchozí hlas na novém úderu.
    """
    if not pedals:
        return notes
    out = []
    for n in notes:
        end = n.end_beat
        for down, up in pedals:
            if down <= end < up:
                end = up
                break
        out.append(Note(n.pitch, n.start_beat, end - n.start_beat, n.velocity,
                        n.glide))
    last: dict = {}
    for n in out:
        prev = last.get(n.pitch)
        if prev is not None and prev.end_beat > n.start_beat:
            prev.dur_beats = max(1e-3, n.start_beat - prev.start_beat)
        last[n.pitch] = n
    return out


def _sampled_events(track: Track) -> list[list]:
    """Noty samplované stopy -> `[start, dur, vel, pitch, legato_in, legato_out]`.

    Legato: artikulace ho umí (`Sampler.legato`), stopa je mono nebo nota
    má `>`, a nota začíná nejpozději `LEGATO_GAP` po konci předchozí.
    Předchozí nota pak končí na začátku nové a prolnou se.
    """
    notes = _pedaled(track.all_notes(), getattr(track, "pedals", []))
    events = [[n.start_beat, n.dur_beats, n.velocity, n.pitch, False, False]
              for n in notes]
    if not getattr(track.patch, "legato", False):
        return events
    prev = None
    for ev, n in zip(events, notes):
        if (prev is not None and (track.mono or n.glide)
                and prev[0] < ev[0] <= prev[0] + prev[1] + LEGATO_GAP):
            prev[1] = ev[0] - prev[0]
            prev[5] = True
            ev[4] = True
        prev = ev
    return events


def _render_sampled(track: Track, buf: np.ndarray, ctx: FxContext, cache: dict,
                    beat_offset: float) -> None:
    """Noty samplované stopy -> hlasy sampleru přičtené do `buf`."""
    sampler = track.patch
    sr, n = ctx.sr, buf.shape[0]
    spb = 60.0 / ctx.bpm
    skey = sampler.key()
    for idx, (start_b, dur_b, vel, pitch, leg_in, leg_out) in enumerate(
            _sampled_events(track)):
        start = start_b - beat_offset
        if start < 0:
            continue
        i = int(round(start * spb * sr))
        if i >= n:
            continue
        dur = round(max(0.01, dur_b * spb), 5)
        for region in sampler.select(pitch, vel, idx):
            key = (skey, sampler.region_id(region), pitch, dur, leg_in, leg_out, sr)
            voice = cache.get(key)
            if voice is None:
                voice = sampler.voice(region, pitch, dur, sr, leg_in, leg_out)
                cache[key] = voice
            _add(buf, i, voice, sampler.gain(region, vel))

def f0(x, sr=SR, fmin=40.0, fmax=2000.0):
    """Základní frekvence autokorelací (YIN-lite) — odolné proti oktávám."""
    x = x - x.mean()
    n = x.size
    size = 1 << (2 * n - 1).bit_length()
    spec = np.fft.rfft(x, size)
    acf = np.fft.irfft(spec * np.conj(spec), size)[: int(sr / fmin) + 1]
    energy = np.concatenate([[0.0], np.cumsum(x ** 2)])
    taus = np.arange(acf.size)
    d = energy[n] + (energy[n] - energy[taus]) - 2 * acf
    d[0] = 0.0
    cmnd = np.ones_like(d)
    cmnd[1:] = d[1:] * taus[1:] / np.maximum(np.cumsum(d[1:]), 1e-12)
    lo = int(sr / fmax)
    below = np.nonzero(cmnd[lo:] < 0.15)[0]
    t = lo + (int(below[0]) if below.size else int(np.argmin(cmnd[lo:])))
    while t + 1 < cmnd.size and cmnd[t + 1] < cmnd[t]:
        t += 1
    return sr / t
